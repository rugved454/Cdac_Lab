Common Table Expressions::
1. Write CTE having recursive function which display first 15 even numbers.
ans:
		WITH Recursive even_no as
		(
			select 2 as n , 1 as cnt
			union all 
			select n+2,cnt+1
			from even_no
			where cnt<15
		
		)
		select n from even_no;

2. Write a query to find the average salary for each department and then list the names of
employees who earn more than the avg sal of their respective departments.
ans:
		With average as 
		(
			select deptno,avg(sal) av
			from emp 
			group by deptno
		)
		select e.ename,e.sal,e.deptno
		from emp e
		join average avg on e.deptno=av.deptno
		where e.sal>avg.av;


3. Write a query to display empno, ename, deptno and sum(sal)  using CTE.
ans:
		With display_emp as
		(
		select e.ename,e.deptno,d.dname
		from emp e
		join dept d
		on e.deptno=d.deptno
		)
		
		select sum(e.sal) 
		from display_emp e1
		join emp e
		on e1.deptno=e.deptno;		
	

4. Write a query to display top 2 departments having avg salay. Display deptno, dname and avg sal.
ans:
		With top_2 as
		(
			select d.deptno, d.dname, avg(e.sal) as avg_sal
			from dept d
			join emp e on e.deptno=d.deptno
			group by d.deptno,d.dname
		)
		select deptno,dname,avg_sal
		from top_2
		order by avg_sal desc
		limit 2;
		
		
Cursors::
1. Write a cursor to increment salary of all employees by 5% if
employee has joined in year 1981.
ans:
		delimiter //
		create procedure increment_sal()
		begin
			declare emp_id int;
			declare emp_sal decimal(7,2);
			declare emp_date date;
			declare f int default 0;
			
			declare cur cursor for
				select empno,sal,hiredate from emp;
			declare continue handler for not found set f=1;
			open cur;
			x: loop
				fetch cur into emp_id,emp_sal,emp_date;
				if f=1 then 
					leave x;
				end if ;
				
				if year(emp_date)=1981 then 
					update emp
					set sal=sal+(sal*0.05)
					where empno=emp_id;
				end if ;
				end loop;
				close cur;
			end //
			delimiter ;
