2.	Write the procedure to accept job as in parameter and perform following actions.
Hike in sal by  5% if employee is manager.
Hike in sal by 10% if employee is clerk
Hike in sal by 7% if employee is analyse
otherwise hike by 3%.
Make necessary changes in emp_data  table.
ans:

	delimiter //
	drop procedure if exists hike_job;
	create procedure hike_job(in p_job varchar(9))
	begin
	declare v_multiplier decimal(9,2);
	if(p_job = 'MANAGER') then  set v_multiplier = 1.05;
	elseif(p_job = 'CLERK') then set v_multiplier = 1.1; 
	elseif(p_job = 'ANALYST') then set v_multiplier = 1.07;
	else set v_multiplier = 1.03;
	end if;
	update emp_data 
	set sal = sal*v_multiplier where job = p_job;
	end //
	delimiter ;
	
	call hike_job('ANALYST');

1. write a function to calculate total salary of an employee. call the function in select clause.
total sal=sal+ comm
ans:
delimiter //
drop function if exists total_salary //
create function total_salary(p_empno int) returns decimal(7,2)
deterministic
begin
    declare v_sal decimal(7,2);
    declare v_comm decimal(7,2);
    select sal, ifnull(comm, 0) into v_sal, v_comm from emp where empno = p_empno;
    return v_sal + v_comm;
end //
delimiter ;
-- call in select: select empno, ename, total_salary(empno) as total_sal from emp;

3. write a function which returns a factorial of a number to the procedure 'facto'. check if number is positive. print the factorial of a number outside the procedure. pass number as a in parameter to the procedure. call function to calculate factorial of a number in the procedure. function should return factorial to the procedure and procedure should display number and the factorial out of procedure.
ans:
delimiter //
drop function if exists factorial //
create function factorial(p_num int) returns int
deterministic
begin
    declare v_result int default 1;
    declare i int default 1;
    if p_num < 0 then
        signal sqlstate '45000' set message_text = 'number must be positive';
    end if;
    while i <= p_num do
        set v_result = v_result * i;
        set i = i + 1;
    end while;
    return v_result;
end //
delimiter ;

delimiter //
drop procedure if exists facto //
create procedure facto(in p_num int)
begin
    declare v_fact int;
    set v_fact = factorial(p_num);
    select p_num as number, v_fact as factorial;
end //
delimiter ;
-- call: call facto(5);

4. write a query to display maximum salary, minimum salary and average sal along with emp name, deptno and dname and location for all departments.
ans:
select e.ename, e.deptno, d.dname, d.loc,
       max(e.sal) over (partition by e.deptno) as max_sal,
       min(e.sal) over (partition by e.deptno) as min_sal,
       avg(e.sal) over (partition by e.deptno) as avg_sal
from emp e
join dept d on e.deptno = d.deptno;

from (
