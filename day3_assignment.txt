Grouping Data together (Use emp, dept and salgrade table)
1.	Find how much salary amount is invested in the employees working as a salesman of the company.

ans:
	select sum(sal) from emp where job='salesman';

2.	Find minimum and maximum salary earned by any employee working as a clerk in the company.
ans:
	select max(sal),min(sal)
	from emp where job='clerk';

3.	Find how many employees were hired in the year 1980.
ans:
		select * from emp where year(hiredate)='1980';
		
		
4.	Display  sum of salary and minimum salary for all employees who are managers.
ans: 
		select sum(sal),min(sal) fro emp where job='manager';
		
5.	Find total salary for each dept where there is more than 1 employee working as a clerk .  Use emp table.  
ans:	
select deptno , sum(sal) from emp where job='clerk' group by deptno having count(*)>1;

6.	Calculate how many employees earn sal more than 1500 in each dept. Arrange the data according to the deptno in ascending order.

ans:	
		select deptno,count(*) from emp where sal>1500 group by deptno order by deptno;

	
7.	Display number of clerks  in each department.
ans:	
		select deptno,count(*) from  emp where job='clerk' group by deptno;
8.	List minimum salary for each department.

ans:
		select deptno,min(sal) from emp group by deptno;
		
9.	Calculate total amount invested by company in the salary of employees for each and every department. List deptno and total of salaries if total is more than 10000.
ans:
		select deptno,sum(sal) from emp group by deptno having sum(sal)>10000;
		
		
10.	Display job wise sum of salary if sum salary >5000 for all departments. Excluding dept 30.
ans:

		select deptno,sum(sal) from emp where deptno!=30 group by deptno having sum(sal)>5000 ;

Create following tables with proper constraints
book(bid, bname, lid ,aid)
bid-PK use auto_increment
lid-FK on delete set null
aid-FK on delete set null
author(aid, aname)
aid-PK use auto_increment
language(lid, lname)
lid-PK use auto_increment

ans:

	CREATE TABLE language (
	lid INT AUTO_INCREMENT PRIMARY KEY,
	lname VARCHAR(50) NOT NULL
	);

	CREATE TABLE author (
		aid INT AUTO_INCREMENT PRIMARY KEY,
		aname VARCHAR(100) NOT NULL
	);

	CREATE TABLE book (
		bid INT AUTO_INCREMENT PRIMARY KEY,
		bname VARCHAR(100) NOT NULL,
		lid INT,
		aid INT,
		FOREIGN KEY (lid) REFERENCES language(lid) ON DELETE SET NULL,
		FOREIGN KEY (aid) REFERENCES author(aid) ON DELETE SET NULL
	);
